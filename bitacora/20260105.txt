```typescript
import { PrismaClient } from '@prisma/client';
import 'dotenv/config'; // Carga automática del archivo .env

const prisma = new PrismaClient();

async function main() {
  console.log('>>> Iniciando el sembrado de cargos...');

  const cargosData = [
    {
      nombre: 'Analista comercial de almacen I',
      descripcion: 'Analista comercial de almacen I',
    },
    {
      nombre: 'Supervisor de almacen',
      descripcion: 'Supervisor de almacen',
    },
  ];

  for (const cargo of cargosData) {
    const upserted = await prisma.cargos.upsert({
      where: { nombre: cargo.nombre },
      update: {
        fecha_hora_actualizacion: new Date(), // Hora del sistema
      },
      create: {
        nombre: cargo.nombre,
        descripcion: cargo.descripcion,
        fecha_hora_creacion: new Date(),      // Hora del sistema
        fecha_hora_actualizacion: new Date(), // Hora del sistema
      },
    });
    console.log(`✅ Registro procesado: ${upserted.nombre}`);
  }
}

main()
  .then(async () => {
    console.log('>>> Sembrado completado con éxito.');
    await prisma.$disconnect();
  })
  .catch(async (e) => {
    console.error('❌ Error detectado:', e);
    await prisma.$disconnect();
    process.exit(1);
  });
```


### Opción 1: Usar `npx` (Sin instalación global)

Si quieres probar una herramienta CLI de la comunidad rápidamente sin ensuciar tus dependencias globales, puedes usar `npx`.

1. Abre el terminal (CMD).
2. Ejecuta el siguiente comando (necesitarás una **API KEY** de Google AI Studio):
```cmd
set GOOGLE_API_KEY=tu_clave_aqui
npx @google/generative-ai

```


*Nota: Este comando varía según el paquete específico de la comunidad que elijas.*

---

### Opción 2: Instalar una herramienta CLI mediante NPM

Si planeas usarlo con frecuencia, puedes instalar un paquete globalmente.

1. **Instalación:**
```cmd
npm install -g gemini-chat-cli

```


2. **Configuración de la API Key:**
Para que funcione en MS-DOS, debes establecer la variable de entorno:
```cmd
setx GEMINI_API_KEY "tu_api_key_aqui"

```


*(Debes cerrar y volver a abrir el prompt después de usar `setx`)*.
3. **Ejecución:**
```cmd
gemini-chat

```

### Opción 3: Ejecutar tu propio script de Gemini

Como ya tienes un entorno de **TypeScript** configurado en tu carpeta `api-backend`, puedes crear un pequeño script para interactuar con Gemini directamente.

1. **Instala el SDK oficial:**
```cmd
npm install @google/generative-ai

```


2. **Crea un archivo llamado `gemini-test.ts**` en tu carpeta `src` con este contenido básico:
```typescript
import { GoogleGenerativeAI } from "@google/generative-ai";
import 'dotenv/config';

const genAI = new GoogleGenerativeAI(process.env.GEMINI_API_KEY!);
const model = genAI.getGenerativeModel({ model: "gemini-pro" });

async function run() {
  const prompt = "Hola, ¿estás funcionando desde mi terminal?";
  const result = await model.generateContent(prompt);
  console.log(result.response.text());
}
run();

```


3. **Ejecútalo desde el prompt** usando `tsx` (que ya tienes en tu lista de dependencias):
```cmd
npx tsx src/gemini-test.ts

```

### Consideraciones para MS-DOS (CMD)

* **Variables de Entorno:** Recuerda que en el CMD tradicional se usa `set` para la sesión actual y `setx` para cambios permanentes.
* **Permisos:** Si recibes errores al instalar con `-g`, asegúrate de ejecutar el Prompt de Comando como **Administrador**.
* **UTF-8:** Si los caracteres especiales (tildes, eñes) se ven mal en el prompt, ejecuta `chcp 65001` antes de iniciar tu CLI para activar el soporte de UTF-8.
